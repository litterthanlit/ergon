# 3D Creator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the 2D canvas MeshRenderer with a cinematic Three.js WebGL renderer featuring 7 custom GLSL render modes, post-processing pipeline, mixed media image planes, adaptive quality, and parallax/orbit camera.

**Architecture:** R3F (React Three Fiber) manages the scene graph, render loop, and canvas lifecycle. Raw Three.js ShaderMaterial with hand-written GLSL handles all visual rendering. A shared `core.glsl` provides noise functions and palette utilities; each of the 7 render modes gets its own vertex/fragment shader pair. Post-processing is 7 independently toggleable effects via `@react-three/postprocessing`.

**Tech Stack:** React 19, Next.js 16, TypeScript 5 (strict), Three.js, @react-three/fiber, @react-three/drei, @react-three/postprocessing, detect-gpu, Zustand 5, Vitest 4

```typescript
import { create } from "zustand";

export type GridPoint = {
  id: string;
  col: number;
  row: number;
  worldX: number;
  worldY: number;
};

export type Edge = {
  from: string;
  to: string;
};

export type RenderMode = "fluid" | "nebula" | "crystal" | "mycelium" | "plasma" | "erosion" | "flow";

export const RENDER_MODES: { id: RenderMode; label: string }[] = [
  { id: "fluid", label: "Fluid" },
  { id: "nebula", label: "Nebula" },
  { id: "crystal", label: "Crystal" },
  { id: "mycelium", label: "Mycelium" },
  { id: "plasma", label: "Plasma" },
  { id: "erosion", label: "Erosion" },
  { id: "flow", label: "Flow" },
];

export type PostFX = {
  bloom: boolean;
  chromatic: boolean;
  vignette: boolean;
  dof: boolean;
  grain: boolean;
  toneMapping: boolean;
  motionBlur: boolean;
};

export type ImagePlane = {
  id: string;
  url: string;
  position: [number, number, number];
  rotation: [number, number, number];
  scale: number;
};

type CreatorState = {
  // Grid
  gridCols: number;
  gridRows: number;
  points: GridPoint[];
  edges: Edge[];
  selectedPoint: string | null;

  // Animation
  renderMode: RenderMode;
  breathe: number;
  pulseSpeed: number;
  extrudeDepth: number;

  // Drivers
  palette: string[];
  tempo: number;
  seed: number;

  // Post-processing
  postFX: PostFX;

  // Image planes
  imagePlanes: ImagePlane[];

  // Actions
  initGrid: (cols: number, rows: number, width: number, height: number) => void;
  selectPoint: (id: string) => void;
  clearSelection: () => void;
  addEdge: (from: string, to: string) => void;
  removeLastEdge: () => void;
  clearAll: () => void;
  setRenderMode: (mode: RenderMode) => void;
  setBreathe: (v: number) => void;
  setPulseSpeed: (v: number) => void;
  setExtrudeDepth: (v: number) => void;
  setPalette: (colors: string[]) => void;
  setTempo: (v: number) => void;
  randomizeSeed: () => void;
  togglePostFX: (effect: keyof PostFX) => void;
  addImagePlane: (url: string, position: [number, number, number]) => void;
  removeImagePlane: (id: string) => void;
  updateImagePlane: (id: string, transform: Partial<Pick<ImagePlane, "position" | "rotation" | "scale">>) => void;
};

function hasEdge(edges: Edge[], from: string, to: string): boolean {
  return edges.some(
    (e) => (e.from === from && e.to === to) || (e.from === to && e.to === from)
  );
}

let planeIdCounter = 0;

export const useCreatorStore = create<CreatorState>((set) => ({
  gridCols: 16,
  gridRows: 10,
  points: [],
  edges: [],
  selectedPoint: null,

  renderMode: "fluid" as RenderMode,
  breathe: 8,
  pulseSpeed: 1.0,
  extrudeDepth: 0,

  palette: ["#00ffa3", "#0088ff", "#cc44ff", "#ffffff", "#ff2d6b"],
  tempo: 1.0,
  seed: Math.floor(Math.random() * 10000),

  postFX: {
    bloom: false,
    chromatic: false,
    vignette: false,
    dof: false,
    grain: false,
    toneMapping: false,
    motionBlur: false,
  },

  imagePlanes: [],

  initGrid: (cols, rows, width, height) => {
    const spacingX = width / (cols + 1);
    const spacingY = height / (rows + 1);
    const points: GridPoint[] = [];
    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        points.push({
          id: `${c}-${r}`,
          col: c,
          row: r,
          worldX: (c + 1) * spacingX,
          worldY: (r + 1) * spacingY,
        });
      }
    }
    set({ points, gridCols: cols, gridRows: rows, edges: [], selectedPoint: null });
  },

  selectPoint: (id) =>
    set((state) => {
      if (!state.selectedPoint) {
        return { selectedPoint: id };
      }
      if (state.selectedPoint === id) {
        return { selectedPoint: null };
      }
      const from = state.selectedPoint;
      const to = id;
      if (hasEdge(state.edges, from, to)) {
        return { selectedPoint: id };
      }
      return {
        edges: [...state.edges, { from, to }],
        selectedPoint: id,
      };
    }),

  clearSelection: () => set({ selectedPoint: null }),

  addEdge: (from, to) =>
    set((state) => {
      if (hasEdge(state.edges, from, to)) return state;
      return { edges: [...state.edges, { from, to }] };
    }),

  removeLastEdge: () =>
    set((state) => ({
      edges: state.edges.slice(0, -1),
    })),

  clearAll: () => set({ edges: [], selectedPoint: null, imagePlanes: [] }),
  setRenderMode: (mode) => set({ renderMode: mode }),

  setBreathe: (v) => set({ breathe: v }),
  setPulseSpeed: (v) => set({ pulseSpeed: v }),
  setExtrudeDepth: (v) => set({ extrudeDepth: v }),
  setPalette: (colors) => set({ palette: colors }),
  setTempo: (v) => set({ tempo: v }),
  randomizeSeed: () => set({ seed: Math.floor(Math.random() * 10000) }),

  togglePostFX: (effect) =>
    set((state) => ({
      postFX: { ...state.postFX, [effect]: !state.postFX[effect] },
    })),

  addImagePlane: (url, position) =>
    set((state) => {
      if (state.imagePlanes.length >= 10) return state;
      const plane: ImagePlane = {
        id: `plane-${++planeIdCounter}`,
        url,
        position,
        rotation: [0, 0, 0],
        scale: 1,
      };
      return { imagePlanes: [...state.imagePlanes, plane] };
    }),

  removeImagePlane: (id) =>
    set((state) => ({
      imagePlanes: state.imagePlanes.filter((p) => p.id !== id),
    })),

  updateImagePlane: (id, transform) =>
    set((state) => ({
      imagePlanes: state.imagePlanes.map((p) =>
        p.id === id ? { ...p, ...transform } : p
      ),
    })),
}));
```

```glsl
// ============================================================
// core.glsl — Shared noise, palette, and utility functions
// Included at the top of every render mode's vertex/fragment shader
// ============================================================

// --- Common uniforms (declared in each ShaderMaterial, values from store) ---
// uniform float uTime;
// uniform float uBreathe;
// uniform float uPulseSpeed;
// uniform float uTempo;
// uniform float uSeed;
// uniform vec3 uPalette[5];

// ============================================================
// SIMPLEX NOISE 3D (Ashima Arts / Ian McEwan)
// ============================================================

vec3 mod289(vec3 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 mod289(vec4 x) { return x - floor(x * (1.0 / 289.0)) * 289.0; }
vec4 permute(vec4 x) { return mod289(((x * 34.0) + 10.0) * x); }
vec4 taylorInvSqrt(vec4 r) { return 1.79284291400159 - 0.85373472095314 * r; }

float snoise(vec3 v) {
  const vec2 C = vec2(1.0 / 6.0, 1.0 / 3.0);
  const vec4 D = vec4(0.0, 0.5, 1.0, 2.0);

  vec3 i = floor(v + dot(v, C.yyy));
  vec3 x0 = v - i + dot(i, C.xxx);

  vec3 g = step(x0.yzx, x0.xyz);
  vec3 l = 1.0 - g;
  vec3 i1 = min(g.xyz, l.zxy);
  vec3 i2 = max(g.xyz, l.zxy);

  vec3 x1 = x0 - i1 + C.xxx;
  vec3 x2 = x0 - i2 + C.yyy;
  vec3 x3 = x0 - D.yyy;

  i = mod289(i);
  vec4 p = permute(permute(permute(
    i.z + vec4(0.0, i1.z, i2.z, 1.0))
    + i.y + vec4(0.0, i1.y, i2.y, 1.0))
    + i.x + vec4(0.0, i1.x, i2.x, 1.0));

  float n_ = 0.142857142857;
  vec3 ns = n_ * D.wyz - D.xzx;

  vec4 j = p - 49.0 * floor(p * ns.z * ns.z);

  vec4 x_ = floor(j * ns.z);
  vec4 y_ = floor(j - 7.0 * x_);

  vec4 x = x_ * ns.x + ns.yyyy;
  vec4 y = y_ * ns.x + ns.yyyy;
  vec4 h = 1.0 - abs(x) - abs(y);

  vec4 b0 = vec4(x.xy, y.xy);
  vec4 b1 = vec4(x.zw, y.zw);

  vec4 s0 = floor(b0) * 2.0 + 1.0;
  vec4 s1 = floor(b1) * 2.0 + 1.0;
  vec4 sh = -step(h, vec4(0.0));

  vec4 a0 = b0.xzyw + s0.xzyw * sh.xxyy;
  vec4 a1 = b1.xzyw + s1.xzyw * sh.zzww;

  vec3 p0 = vec3(a0.xy, h.x);
  vec3 p1 = vec3(a0.zw, h.y);
  vec3 p2 = vec3(a1.xy, h.z);
  vec3 p3 = vec3(a1.zw, h.w);

  vec4 norm = taylorInvSqrt(vec4(dot(p0, p0), dot(p1, p1), dot(p2, p2), dot(p3, p3)));
  p0 *= norm.x;
  p1 *= norm.y;
  p2 *= norm.z;
  p3 *= norm.w;

  vec4 m = max(0.5 - vec4(dot(x0, x0), dot(x1, x1), dot(x2, x2), dot(x3, x3)), 0.0);
  m = m * m;
  return 105.0 * dot(m * m, vec4(dot(p0, x0), dot(p1, x1), dot(p2, x2), dot(p3, x3)));
}

// ============================================================
// CURL NOISE (derived from simplex)
// ============================================================

vec3 curlNoise(vec3 p) {
  float e = 0.1;
  vec3 dx = vec3(e, 0.0, 0.0);
  vec3 dy = vec3(0.0, e, 0.0);
  vec3 dz = vec3(0.0, 0.0, e);

  float n_y_dz = snoise(p + dz);
  float n_y_mdz = snoise(p - dz);
  float n_z_dy = snoise(p + dy);
  float n_z_mdy = snoise(p - dy);
  float n_x_dz = snoise(p + dz + vec3(31.416, 0.0, 0.0));
  float n_x_mdz = snoise(p - dz + vec3(31.416, 0.0, 0.0));
  float n_z_dx = snoise(p + dx + vec3(0.0, 31.416, 0.0));
  float n_z_mdx = snoise(p - dx + vec3(0.0, 31.416, 0.0));
  float n_x_dy = snoise(p + dy + vec3(0.0, 0.0, 31.416));
  float n_x_mdy = snoise(p - dy + vec3(0.0, 0.0, 31.416));

  float curl_x = (n_z_dy - n_z_mdy - n_y_dz + n_y_mdz) / (2.0 * e);
  float curl_y = (n_x_dz - n_x_mdz - n_z_dx + n_z_mdx) / (2.0 * e);
  float curl_z = (n_y_dz - n_y_mdz - n_x_dy + n_x_mdy) / (2.0 * e);

  return vec3(curl_x, curl_y, curl_z);
}

// ============================================================
// WORLEY NOISE (cellular / Voronoi)
// ============================================================

vec2 worley(vec3 p) {
  vec3 i_st = floor(p);
  vec3 f_st = fract(p);

  float minDist = 1.0;
  float secondMinDist = 1.0;

  for (int z = -1; z <= 1; z++) {
    for (int y = -1; y <= 1; y++) {
      for (int x = -1; x <= 1; x++) {
        vec3 neighbor = vec3(float(x), float(y), float(z));
        vec3 point = vec3(
          fract(sin(dot(i_st + neighbor, vec3(12.9898, 78.233, 45.164))) * 43758.5453),
          fract(sin(dot(i_st + neighbor, vec3(93.9898, 67.345, 12.789))) * 28461.6432),
          fract(sin(dot(i_st + neighbor, vec3(43.321, 94.532, 71.984))) * 63728.1947)
        );
        vec3 diff = neighbor + point - f_st;
        float dist = length(diff);
        if (dist < minDist) {
          secondMinDist = minDist;
          minDist = dist;
        } else if (dist < secondMinDist) {
          secondMinDist = dist;
        }
      }
    }
  }

  return vec2(minDist, secondMinDist);
}

// ============================================================
// FRACTIONAL BROWNIAN MOTION
// ============================================================

float fbm(vec3 p, int octaves) {
  float value = 0.0;
  float amplitude = 0.5;
  float frequency = 1.0;
  for (int i = 0; i < 8; i++) {
    if (i >= octaves) break;
    value += amplitude * snoise(p * frequency);
    frequency *= 2.0;
    amplitude *= 0.5;
  }
  return value;
}

// ============================================================
// PALETTE LOOKUP
// ============================================================

vec3 paletteLookup(float t, vec3 palette[5]) {
  float idx = t * 4.0;
  int i = int(floor(idx));
  float f = fract(idx);
  i = clamp(i, 0, 3);
  return mix(palette[i], palette[i + 1], f);
}

// ============================================================
// SDF PRIMITIVES (for Fluid mode and others)
// ============================================================

float sdSphere(vec3 p, float r) {
  return length(p) - r;
}

float sdCapsule(vec3 p, vec3 a, vec3 b, float r) {
  vec3 ab = b - a;
  vec3 ap = p - a;
  float t = clamp(dot(ap, ab) / dot(ab, ab), 0.0, 1.0);
  return length(p - (a + t * ab)) - r;
}

float smin(float a, float b, float k) {
  float h = clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0);
  return mix(b, a, h) - k * h * (1.0 - h);
}

// ============================================================
// ROTATION MATRIX
// ============================================================

mat3 rotateY(float angle) {
  float c = cos(angle);
  float s = sin(angle);
  return mat3(c, 0, s, 0, 1, 0, -s, 0, c);
}

mat3 rotateX(float angle) {
  float c = cos(angle);
  float s = sin(angle);
  return mat3(1, 0, 0, 0, c, -s, 0, s, c);
}
```

```tsx
"use client";

import { useEffect, useRef, createContext, useContext, useState } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import { getGPUTier, type TierResult } from "detect-gpu";

type QualityTier = 1 | 2 | 3;

type QualitySettings = {
  tier: QualityTier;
  particleMultiplier: number;
  postProcessingScale: number;
  maxParticles: number;
};

const TIER_SETTINGS: Record<QualityTier, QualitySettings> = {
  1: { tier: 1, particleMultiplier: 0.3, postProcessingScale: 0.5, maxParticles: 15000 },
  2: { tier: 2, particleMultiplier: 1.0, postProcessingScale: 1.0, maxParticles: 50000 },
  3: { tier: 3, particleMultiplier: 1.5, postProcessingScale: 1.0, maxParticles: 80000 },
};

const QualityContext = createContext<QualitySettings>(TIER_SETTINGS[2]);

export function useQuality(): QualitySettings {
  return useContext(QualityContext);
}

export function AdaptiveQuality({ children }: { children: React.ReactNode }) {
  const [tier, setTier] = useState<QualityTier>(2);
  const { gl } = useThree();

  // Frame time tracking for adaptive downgrade
  const frameTimes = useRef<number[]>([]);
  const lastTime = useRef(performance.now());
  const stableFrames = useRef(0);
  const currentTier = useRef<QualityTier>(2);
  const baseTier = useRef<QualityTier>(2);

  // Detect GPU tier on mount
  useEffect(() => {
    getGPUTier({ glContext: gl.getContext() as WebGL2RenderingContext }).then((result: TierResult) => {
      let detected: QualityTier = 2;
      if (result.tier <= 1) detected = 1;
      else if (result.tier >= 3) detected = 3;
      baseTier.current = detected;
      currentTier.current = detected;
      setTier(detected);
    });
  }, [gl]);

  // Monitor framerate and adapt
  useFrame(() => {
    const now = performance.now();
    const delta = now - lastTime.current;
    lastTime.current = now;

    frameTimes.current.push(delta);
    if (frameTimes.current.length > 60) frameTimes.current.shift();

    // Check every 30 frames
    if (frameTimes.current.length >= 30 && frameTimes.current.length % 30 === 0) {
      const avg = frameTimes.current.reduce((a, b) => a + b, 0) / frameTimes.current.length;

      if (avg > 20 && currentTier.current > 1) {
        // Dropping below 50fps — downgrade
        currentTier.current = (currentTier.current - 1) as QualityTier;
        setTier(currentTier.current);
        stableFrames.current = 0;
      } else if (avg < 14 && currentTier.current < baseTier.current) {
        // Running well above 60fps — try upgrading
        stableFrames.current++;
        if (stableFrames.current >= 2) {
          currentTier.current = (currentTier.current + 1) as QualityTier;
          setTier(currentTier.current);
          stableFrames.current = 0;
        }
      } else {
        stableFrames.current = 0;
      }
    }
  });

  return (
    <QualityContext.Provider value={TIER_SETTINGS[tier]}>
      {children}
    </QualityContext.Provider>
  );
}
```

```glsl
uniform float uBreathe;
uniform float uPulseSpeed;
uniform float uSeed;
uniform vec3 uPalette[5];

varying vec3 vPosition;
varying vec3 vNormal;
varying float vNoise;

void main() {
  // Fresnel for edge glow
  vec3 viewDir = normalize(cameraPosition - vPosition);
  float fresnel = pow(1.0 - abs(dot(viewDir, vNormal)), 3.0);

  // Worley noise for caustic pattern
  vec2 w = worley(vPosition * 0.02 + vec3(uTime * uPulseSpeed * 0.1));
  float caustic = smoothstep(0.0, 0.3, w.y - w.x);

  // Palette color based on noise
  float colorIdx = vNoise * 0.5 + 0.5;
  vec3 baseColor = paletteLookup(colorIdx, uPalette);

  // Subsurface scattering approximation
  float sss = pow(max(0.0, dot(viewDir, -vNormal)), 2.0) * 0.3;

  // Combine
  vec3 color = baseColor * (0.5 + caustic * 0.5);
  color += fresnel * baseColor * 1.5;
  color += sss * baseColor;

  // Emissive glow for bloom to pick up
  float emissive = fresnel * 0.8 + caustic * 0.3;

  gl_FragColor = vec4(color, 0.9);
}
```

```tsx
"use client";

import { useRef, useMemo } from "react";

function buildShader(mode: string, core: string): string {
  return core + "\n" + mode;
}

type ShaderPair = { vert: string; frag: string };

const SHADERS: Record<RenderMode, ShaderPair> = {
  fluid: { vert: buildShader(fluidVert, coreGlsl), frag: buildShader(fluidFrag, coreGlsl) },
  // Remaining modes will be added in Tasks 10-15

  crystal: { vert: buildShader(fluidVert, coreGlsl), frag: buildShader(fluidFrag, coreGlsl) },
  mycelium: { vert: buildShader(fluidVert, coreGlsl), frag: buildShader(fluidFrag, coreGlsl) },
  plasma: { vert: buildShader(fluidVert, coreGlsl), frag: buildShader(fluidFrag, coreGlsl) },
  erosion: { vert: buildShader(fluidVert, coreGlsl), frag: buildShader(fluidFrag, coreGlsl) },
```

```tsx
  const velocities = new Float32Array(totalParticles * 3);
  const progress = new Float32Array(totalParticles);
  const pointMap = new Map(points.map((p) => [p.id, p]));

  let idx = 0;
  edges.forEach((edge) => {
    const from = pointMap.get(edge.from);
    const to = pointMap.get(edge.to);
    if (!from || !to) return;

    const fx = (from.worldX / window.innerWidth - 0.5) * 500;
    const fy = -(from.worldY / window.innerHeight - 0.5) * 300;
    const tx = (to.worldX / window.innerWidth - 0.5) * 500;
    const ty = -(to.worldY / window.innerHeight - 0.5) * 300;

    const dx = tx - fx;
    const dy = ty - fy;

    for (let p = 0; p < particlesPerEdge; p++) {
      const t = p / particlesPerEdge;
      const i3 = idx * 3;
      positions[i3] = fx + dx * t + (Math.random() - 0.5) * 10;
      positions[i3 + 1] = fy + dy * t + (Math.random() - 0.5) * 10;
      positions[i3 + 2] = (Math.random() - 0.5) * 20;
      velocities[i3] = dx * 0.01;
      velocities[i3 + 1] = dy * 0.01;
      velocities[i3 + 2] = (Math.random() - 0.5) * 0.01;
      progress[idx] = Math.random();
      idx++;
    }
  });

  return { positions, velocities, progress, count: idx };
}, [renderMode, edges, points, quality.particleMultiplier]);

// In the return JSX, add a conditional branch:
// Before the existing connectedPositions map:
if (renderMode === "flow" && flowParticles) {
  return (
    <group ref={groupRef}>
      <points>
        <bufferGeometry>
          <bufferAttribute attach="attributes-position" array={flowParticles.positions} count={flowParticles.count} itemSize={3} />
          <bufferAttribute attach="attributes-aVelocity" array={flowParticles.velocities} count={flowParticles.count} itemSize={3} />
          <bufferAttribute attach="attributes-aProgress" array={flowParticles.progress} count={flowParticles.count} itemSize={1} />
        </bufferGeometry>
        <shaderMaterial
          ref={materialRef}
          vertexShader={shaderPair.vert}
          fragmentShader={shaderPair.frag}
          uniforms={makeUniforms(palette)}
          transparent
          depthWrite={false}
          blending={THREE.AdditiveBlending}
        />
      </points>
      {/* Thin edge lines */}
      {edgePositions.length > 0 && (
        <lineSegments>
          <bufferGeometry>
            <bufferAttribute
              attach="attributes-position"
              array={new Float32Array(edgePositions.flatMap((v) => [v.x, v.y, v.z]))}
              count={edgePositions.length}
              itemSize={3}
            />
          </bufferGeometry>
          <lineBasicMaterial color="#ffffff" transparent opacity={0.08} linewidth={1} />
        </lineSegments>
      )}
    </group>
  );
}
```

```tsx
"use client";

import { useRef, useMemo, useCallback, useEffect } from "react";
import { useFrame, useThree } from "@react-three/fiber";
import * as THREE from "three";
import { useCreatorStore } from "@/lib/creator-store";

// Simple hash for deterministic randomness from seed
function seededRandom(seed: number, index: number): number {
  const x = Math.sin(seed * 12.9898 + index * 78.233) * 43758.5453;
  return x - Math.floor(x);
}

// Generate procedural texture on a canvas
function generateTexture(seed: number, index: number, palette: string[]): THREE.CanvasTexture {
  const size = 128;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;
  const ctx = canvas.getContext("2d")!;

  const type = index % 4;
  const color = palette[index % palette.length];

  if (type === 0) {
    // Noise gradient
    const gradient = ctx.createLinearGradient(0, 0, size, size);
    gradient.addColorStop(0, color + "40");
    gradient.addColorStop(1, palette[(index + 1) % palette.length] + "30");
    ctx.fillStyle = gradient;
    ctx.fillRect(0, 0, size, size);
  } else if (type === 1) {
    // Voronoi-like cells
    ctx.fillStyle = "#00000000";
    ctx.fillRect(0, 0, size, size);
    for (let i = 0; i < 8; i++) {
      const x = seededRandom(seed, index * 100 + i) * size;
      const y = seededRandom(seed, index * 100 + i + 50) * size;
      const r = 10 + seededRandom(seed, index * 100 + i + 100) * 30;
      ctx.beginPath();
      ctx.arc(x, y, r, 0, Math.PI * 2);
      ctx.fillStyle = color + "25";
      ctx.fill();
    }
  } else if (type === 2) {
    // Glitch horizontal lines
    for (let y = 0; y < size; y += 3) {
      ctx.fillStyle = seededRandom(seed, index * 200 + y) > 0.5 ? color + "15" : "#00000000";
      ctx.fillRect(0, y, size, 2);
    }
  } else {
    // Color field wash
    ctx.fillStyle = color + "20";
    ctx.fillRect(0, 0, size, size);
    ctx.filter = "blur(20px)";
    ctx.fillStyle = palette[(index + 2) % palette.length] + "30";
    ctx.beginPath();
    ctx.arc(size * 0.3, size * 0.4, size * 0.3, 0, Math.PI * 2);
    ctx.fill();
  }

  const texture = new THREE.CanvasTexture(canvas);
  texture.needsUpdate = true;
  return texture;
}

function AutoPlane({ index, seed, palette, edgeCount }: {
  index: number;
  seed: number;
  palette: string[];
  edgeCount: number;
}) {
  const meshRef = useRef<THREE.Mesh>(null);

  const texture = useMemo(() => generateTexture(seed, index, palette), [seed, index, palette]);

  const position = useMemo((): [number, number, number] => {
    const x = (seededRandom(seed, index * 3) - 0.5) * 400;
    const y = (seededRandom(seed, index * 3 + 1) - 0.5) * 250;
    const z = (seededRandom(seed, index * 3 + 2) - 0.5) * 100 + (index % 2 === 0 ? 50 : -50);
    return [x, y, z];
  }, [seed, index]);

  const scale = 30 + seededRandom(seed, index * 7) * 40;

  useFrame((_state, delta) => {
    if (!meshRef.current) return;
    // Slow drift and rotation
    meshRef.current.position.x += Math.sin(performance.now() * 0.0001 + index) * delta * 0.5;
    meshRef.current.position.y += Math.cos(performance.now() * 0.00015 + index) * delta * 0.3;
    meshRef.current.rotation.z += delta * 0.01 * (index % 2 === 0 ? 1 : -1);
  });

  return (
    <mesh ref={meshRef} position={position}>
      <planeGeometry args={[scale, scale * 0.7]} />
      <meshBasicMaterial
        map={texture}
        transparent
        opacity={0.4}
        side={THREE.DoubleSide}
        depthWrite={false}
        blending={THREE.AdditiveBlending}
      />
    </mesh>
  );
}

function UserPlane({ plane }: { plane: { id: string; url: string; position: [number, number, number]; rotation: [number, number, number]; scale: number } }) {
  const meshRef = useRef<THREE.Mesh>(null);

  const texture = useMemo(() => {
    const tex = new THREE.TextureLoader().load(plane.url);
    return tex;
  }, [plane.url]);

  return (
    <mesh
      ref={meshRef}
      position={plane.position}
      rotation={plane.rotation}
      scale={plane.scale}
    >
      <planeGeometry args={[40, 30]} />
      <meshBasicMaterial
        map={texture}
        transparent
        opacity={0.85}
        side={THREE.DoubleSide}
        depthWrite={false}
      />
    </mesh>
  );
}

export function ImagePlanes() {
  const edges = useCreatorStore((s) => s.edges);
  const seed = useCreatorStore((s) => s.seed);
  const palette = useCreatorStore((s) => s.palette);
  const imagePlanes = useCreatorStore((s) => s.imagePlanes);
  const addImagePlane = useCreatorStore((s) => s.addImagePlane);
  const { gl } = useThree();

  // Auto-generated plane count: 0 when no edges, scales with edge count
  const autoPlaneCount = edges.length === 0 ? 0 : Math.min(2 + Math.floor(edges.length / 3), 4);

  // Drag-drop handler
  useEffect(() => {
    const canvas = gl.domElement;

    const handleDragOver = (e: DragEvent) => {
      e.preventDefault();
      e.stopPropagation();
    };

    const handleDrop = (e: DragEvent) => {
      e.preventDefault();
      e.stopPropagation();
      const files = e.dataTransfer?.files;
      if (!files) return;

      for (let i = 0; i < files.length; i++) {
        const file = files[i];
        if (!file.type.startsWith("image/")) continue;
        const url = URL.createObjectURL(file);
        const x = (e.clientX / window.innerWidth - 0.5) * 500;
        const y = -(e.clientY / window.innerHeight - 0.5) * 300;
        const z = (Math.random() - 0.5) * 80;
        addImagePlane(url, [x, y, z]);
      }
    };

    canvas.addEventListener("dragover", handleDragOver);
    canvas.addEventListener("drop", handleDrop);
    return () => {
      canvas.removeEventListener("dragover", handleDragOver);
      canvas.removeEventListener("drop", handleDrop);
    };
  }, [gl, addImagePlane]);

  return (
    <group>
      {/* Auto-generated planes */}
      {Array.from({ length: autoPlaneCount }).map((_, i) => (
        <AutoPlane key={`auto-${seed}-${i}`} index={i} seed={seed} palette={palette} edgeCount={edges.length} />
      ))}

      {/* User-dropped planes */}
      {imagePlanes.map((plane) => (
        <UserPlane key={plane.id} plane={plane} />
      ))}
    </group>
  );
}
```
